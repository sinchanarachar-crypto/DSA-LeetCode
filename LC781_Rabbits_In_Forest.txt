class Solution
{
public:
    int numRabbits(vector<int>& answers)
    {
        unordered_map<int, int> mp;

        for(int &x : answers)
        {
            mp[x]++;
        }

        int total = 0;

        for(auto &it : mp)
        {
            int ans = it.first; // answered rabit
            int count = it.second; // answer said by rabit

            int groupSize = (ans + 1);
            int groups = (ceil)((double)count/groupSize);

            total += groups * groupSize;
        }

        return total;
    }
};
